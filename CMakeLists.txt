cmake_minimum_required(VERSION 3.22)
project(SLXSeaBalance VERSION 0.3.0 LANGUAGES C CXX)
include(cmake/standards.cmake)
include(CTest)
find_package(Threads REQUIRED)

set(platform "")
if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(platform windows-x64)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
  set(platform linux-x64)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SIZEOF_VOID_P EQUAL 4 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^arm")
  set(platform linux-armv7)
endif()
set(SLX_CORE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/core/${platform}" CACHE PATH "Directory with the evaluation core binaries")
set(model_library "${CMAKE_SHARED_LIBRARY_PREFIX}slx_seabalance_model${CMAKE_SHARED_LIBRARY_SUFFIX}")
set(slx_build_pc_default OFF)
if(TARGET slx_seabalance_model OR EXISTS "${SLX_CORE_DIR}/${model_library}")
  set(slx_build_pc_default ON)
endif()

option(SLX_BUILD_PC "Build simulator CLI and support tools" ${slx_build_pc_default})
option(SLX_BUILD_DESKTOP "Build the OpenGL desktop" ${SLX_BUILD_PC})
option(SLX_BUILD_SERVICE "Build the Linux/PC controller service" ON)
option(SLX_BUILD_HMI "Build the 480x272 GTK 3 panel" OFF)
if(SLX_BUILD_DESKTOP AND NOT SLX_BUILD_PC)
  message(FATAL_ERROR "SLX_BUILD_DESKTOP requires SLX_BUILD_PC")
endif()

function(slx_stage_core name required)
  if(TARGET slx_seabalance_${name})
    return()
  endif()
  set(file "${CMAKE_SHARED_LIBRARY_PREFIX}slx_seabalance_${name}${CMAKE_SHARED_LIBRARY_SUFFIX}")
  if(EXISTS "${SLX_CORE_DIR}/${file}")
    file(MAKE_DIRECTORY "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
    configure_file("${SLX_CORE_DIR}/${file}" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/${file}" COPYONLY)
    install(FILES "${SLX_CORE_DIR}/${file}" DESTINATION bin)
  elseif(required)
    message(FATAL_ERROR "Missing SeaBalance core (${platform}): set SLX_CORE_DIR to the directory containing ${file} from the evaluation SDK, or set SLX_BUILD_PC=OFF and SLX_BUILD_DESKTOP=OFF for the controller-only build")
  endif()
endfunction()
if(SLX_BUILD_PC)
  slx_stage_core(model TRUE)
endif()
if(SLX_BUILD_PC OR SLX_BUILD_SERVICE)
  slx_stage_core(control FALSE)
endif()

add_library(slx-sdk STATIC sdk/src/runtime.cpp sdk/src/wire.cpp sdk/src/socket.cpp sdk/src/link.cpp)
target_include_directories(slx-sdk PUBLIC sdk/include sdk/src)
target_link_libraries(slx-sdk PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_compile_definitions(slx-sdk PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(slx-sdk PUBLIC ws2_32)
endif()
slx_warnings(slx-sdk)
add_library(slx-pid STATIC examples/pid.cpp)
target_include_directories(slx-pid PUBLIC examples sdk/include)
slx_warnings(slx-pid)
add_library(slx-json INTERFACE)
target_include_directories(slx-json SYSTEM INTERFACE third_party/include)

if(SLX_BUILD_PC)
  add_library(slx-session STATIC desktop/scenario.cpp desktop/session.cpp desktop/analysis.cpp desktop/run.cpp)
  target_include_directories(slx-session PUBLIC desktop)
  target_link_libraries(slx-session PUBLIC slx-sdk slx-pid slx-json)
  slx_warnings(slx-session)
  add_executable(slx-seabalance-cli desktop/cli.cpp)
  target_link_libraries(slx-seabalance-cli PRIVATE slx-session)
  slx_warnings(slx-seabalance-cli)
  install(TARGETS slx-seabalance-cli RUNTIME DESTINATION bin)
endif()
if(SLX_BUILD_SERVICE)
  add_executable(slx-seabalance-controller embedded/service.cpp embedded/panel_state.cpp)
  target_link_libraries(slx-seabalance-controller PRIVATE slx-sdk slx-pid slx-json)
  slx_warnings(slx-seabalance-controller)
  install(TARGETS slx-seabalance-controller RUNTIME DESTINATION bin)
endif()
if(SLX_BUILD_HMI)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0>=3.22)
  add_executable(slx-seabalance-panel embedded/panel.cpp)
  target_link_libraries(slx-seabalance-panel PRIVATE slx-sdk slx-json PkgConfig::GTK3)
  slx_warnings(slx-seabalance-panel)
  install(TARGETS slx-seabalance-panel RUNTIME DESTINATION bin)
endif()
if(SLX_BUILD_DESKTOP)
  include(cmake/desktop.cmake)
endif()
if(BUILD_TESTING)
  add_executable(slx-contract-tests tests/contract.cpp)
  target_link_libraries(slx-contract-tests PRIVATE slx-sdk slx-pid)
  slx_warnings(slx-contract-tests)
  add_test(NAME sdk-and-wire COMMAND slx-contract-tests)
  if(SLX_BUILD_PC)
    add_executable(slx-simulation-tests tests/simulation.cpp)
    target_link_libraries(slx-simulation-tests PRIVATE slx-session)
    slx_warnings(slx-simulation-tests)
    add_test(NAME simulation-and-replay COMMAND slx-simulation-tests)
    if(SLX_BUILD_SERVICE AND NOT CMAKE_CROSSCOMPILING)
      find_package(Python3 QUIET COMPONENTS Interpreter)
      if(Python3_Interpreter_FOUND)
        add_test(NAME ethernet-loopback COMMAND "${Python3_EXECUTABLE}"
          "${CMAKE_CURRENT_SOURCE_DIR}/tests/loopback.py" "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
        set_tests_properties(ethernet-loopback PROPERTIES TIMEOUT 30)
      endif()
    endif()
  endif()
endif()
install(FILES sdk/include/slx/seabalance.h DESTINATION sdk/include/slx)
install(FILES examples/pid.cpp examples/pid.hpp DESTINATION sdk/examples)
if(SLX_BUILD_PC)
  install(DIRECTORY assets docs licenses DESTINATION bin)
  install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION bin)
endif()
